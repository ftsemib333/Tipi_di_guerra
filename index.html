        let selectedLeft = null;

        let attempts = 0;

        let matchedCount = 0;

 

        function initMatchingGame() {

            const leftCol = document.getElementById('col-left');

            const rightCol = document.getElementById('col-right');

            leftCol.innerHTML = "";

            rightCol.innerHTML = "";

 

            let shuffledRight = [...matchData].sort(() => Math.random() - 0.5);

 

            matchData.forEach((item, idx) => {

                let btnL = document.createElement('button');

                btnL.className = 'match-btn';

                btnL.innerText = item.type;

                btnL.onclick = () => selectMatch('left', idx, btnL);

                leftCol.appendChild(btnL);

            });

 

            shuffledRight.forEach((item) => {

                let btnR = document.createElement('button');

                btnR.className = 'match-btn';

                btnR.innerText = item.desc;

                btnR.onclick = () => selectMatch('right', item.type, btnR);

                rightCol.appendChild(btnR);

            });

        }

 

        function selectMatch(side, val, btn) {

            if (btn.classList.contains('matched')) return;

 

            if (side === 'left') {

                document.querySelectorAll('#col-left .match-btn').forEach(b => b.classList.remove('selected'));

                btn.classList.add('selected');

                selectedLeft = { type: matchData[val].type, btn: btn };

            } else if (side === 'right' && selectedLeft) {

                attempts++;

                document.getElementById('match-attempts').innerText = attempts;

        function renderQuiz() {

            const container = document.getElementById('quiz-container');

            container.innerHTML = "";

            shuffledQuiz = [...rawQuiz].sort(() => Math.random() - 0.5);

 

            shuffledQuiz.forEach((item, index) => {

                let qDiv = document.createElement('div');

                qDiv.className = 'quiz-question';

               

                let qTitle = document.createElement('h4');

                qTitle.innerText = (index + 1) + ". " + item.q;

                qDiv.appendChild(qTitle);

 

                let optsDiv = document.createElement('div');

                optsDiv.className = 'quiz-options';

 

                // Creazione opzioni con indice per shuffle

                let optsWithIdx = item.options.map((opt, i) => ({ text: opt, originalIndex: i }));

                optsWithIdx.sort(() => Math.random() - 0.5);

 

                optsWithIdx.forEach(optObj => {

                    let btn = document.createElement('button');

                    btn.innerText = optObj.text;

                    btn.onclick = () => checkAnswer(index, optObj.originalIndex, btn, optsDiv, item.exp);

                    optsDiv.appendChild(btn);

                });

 

                let fbDiv = document.createElement('div');

                fbDiv.id = 'fb-' + index;

                fbDiv.className = 'feedback-text';

 

                qDiv.appendChild(optsDiv);

                qDiv.appendChild(fbDiv);

                container.appendChild(qDiv);

            });

        }

 

        function checkAnswer(qIdx, selectedOptIdx, clickedBtn, optsContainer, explanation) {

            let buttons = optsContainer.querySelectorAll('button');

            buttons.forEach(b => b.disabled = true);

 

            let correctOptIdx = shuffledQuiz[qIdx].ans;

            let isCorrect = (selectedOptIdx === correctOptIdx);

 

            userQuizAnswers[qIdx] = {

                question: shuffledQuiz[qIdx].q,

                isCorrect: isCorrect,

                correctText: shuffledQuiz[qIdx].options[correctOptIdx]

            };

 

            let fb = document.getElementById('fb-' + qIdx);

 

            if (isCorrect) {

                clickedBtn.classList.add('correct');

                fb.innerText = "Esatto! " + explanation;

                fb.style.color = "#27ae60";

                quizScore++;

            } else {

                clickedBtn.classList.add('wrong');

                fb.innerText = "Errato. " + explanation;

                fb.style.color = "#c0392b";

            }

 

            // Mostra pulsante finale se risposto a tutte

            if (Object.keys(userQuizAnswers).length === 10) {

                document.getElementById('btn-submit-quiz').style.display = 'block';

            }

        }

 

        renderQuiz();

 

        function finishQuiz() {

            document.getElementById('btn-submit-quiz').style.display = 'none';

            document.getElementById('quiz-result').style.display = 'block';

           

            document.getElementById('lbl-score').innerText = "Voto: " + quizScore + " / 10";

 

            let msg = "";

                unlockBadge("Esperto 🏆");

            }

            document.getElementById('lbl-feedback-message').innerText = msg;

        }
